{% trans "Quick Start" %}

{% trans "Three commands to get everything running:" %}

git clone https://github.com/scitex-ai/scitex-hub.git
cd scitex-hub
make start

{% trans "Access at" %} localhost:8000. {% trans "Test user:" %} test-user — {% trans "the password is printed by" %} init_test_user {% trans "on first run, or set" %} SCITEX_HUB_TEST_USER_PASSWORD {% trans "to choose it yourself" %}

{% trans "Prerequisites" %}

  • {% trans "Linux, macOS, or Windows (WSL2)" %}
  • {% trans "Docker 24.0+ with Docker Compose v2" %}
  • {% trans "Git 2.30+" %}
  • {% trans "4 GB RAM minimum (8 GB recommended)" %}

{% trans "Development Setup" %}

  1. {% trans "Clone and configure" %} 
    git clone https://github.com/scitex-ai/scitex-hub.git
cd scitex-hub
cp deployment/docker/envs/.env.example deployment/docker/envs/.env.dev
  2. {% trans "Set secret key" %} ({% trans "edit" %} .env.dev) 
    SCITEX_HUB_DJANGO_SECRET_KEY=$(python3 -c 'import secrets; print(secrets.token_urlsafe(32))')
SCITEX_HUB_POSTGRES_PASSWORD=your-dev-password
  3. {% trans "Start services" %} 
    make start
make status

{% trans "Production Setup" %}

{% trans "For deploying on a home server, NAS, or VPS:" %}

  1. {% trans "Create production environment" %} 
    cp deployment/docker/envs/.env.example deployment/docker/envs/.env.prod
  2. {% trans "Configure for production" %} ({% trans "edit" %} .env.prod) 
    DEBUG=False
SCITEX_HUB_DOMAIN=scitex.example.com
SCITEX_HUB_SITE_URL=https://scitex.example.com
SCITEX_HUB_ENABLE_SSL_REDIRECT=true
  3. {% trans "Start production" %} 
    make ENV=prod start
make ENV=prod status

{% trans "CLI-Only Install" %}

{% trans "Use the CLI and MCP server without Docker:" %}

pip install scitex-hub              # CLI only
pip install scitex-hub[mcp]         # CLI + MCP server
pip install scitex-hub[all]         # Everything

scitex-hub --version
scitex-hub --help

{% trans "MCP Server for AI Agents" %}

{% trans "Connect AI assistants (Claude, Cursor, etc.) to your SciTeX instance:" %}

scitex-hub mcp start                # Start MCP server
scitex-hub mcp doctor               # Diagnose setup
scitex-hub mcp installation          # Client config

make start                    # Start development
make stop                     # Stop all services
make restart                  # Restart services
make status                   # Health check
make logs                     # View all logs
make ENV=prod start           # Start production
make help                     # All available commands

Usage Guide

/* Semantic Tokens - use these in components */
--text-primary           /* Main text */
--text-secondary         /* Secondary text */
--text-muted            /* Muted/disabled text */
--text-inverse          /* Light text on dark backgrounds */

--bg-page               /* Page background */
--bg-surface            /* Card/panel backgrounds */
--bg-muted              /* Subtle surface backgrounds */

--border-default        /* Standard borders */
--border-muted          /* Subtle borders */

--status-success        /* #4a9b7e */
--status-warning        /* #b8956a */
--status-error          /* #a67373 */
--status-info           /* #6b8fb3 */

/* Workspace Colors - Light Theme (uses semantic tokens) */
--bg-page                   /* #fefefe - Main panels */
--bg-surface                /* #fafbfc - Secondary panels */
--bg-muted                  /* #f6f8fa - Hover states */
--border-default            /* #d0d7de - Default borders */
--border-muted              /* #e0e4e8 - Subtle borders */

/* Workspace Colors - Dark Theme */
--workspace-bg-primary      /* #0d0d0d - Main panels */
--workspace-bg-secondary    /* #151515 - Secondary panels */
--workspace-bg-tertiary     /* #1a1a1a - Hover states */
--workspace-bg-elevated     /* #1f1f1f - Elevated surfaces */
--workspace-border-subtle   /* #1a1a1a - Subtle borders */
--workspace-border-default  /* #252525 - Default borders, resizer */
--workspace-border-hover    /* #3a3a3a - Hover state */
--workspace-border-active   /* #4a4a4a - Active/dragging */
--workspace-slider-hover    /* #5a5a5a - Slider handle hover */
--workspace-slider-active   /* #6a6a6a - Slider handle active */

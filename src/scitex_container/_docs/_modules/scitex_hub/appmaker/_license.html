def generate_license_text(
    spdx_id: str, author: str = "", year: int | None = None
) -> str | None:
    """Generate full license text for the given SPDX identifier.

    Parameters
    ----------
    spdx_id : str
        SPDX license identifier (e.g. 'MIT', 'AGPL-3.0').
    author : str
        Copyright holder name.
    year : int, optional
        Copyright year (default: current year).

    Returns
    -------
    str or None
        Filled-in license text, or None if spdx_id is unknown.
    """
    template = LICENSE_TEMPLATES.get(spdx_id)
    if template is None:
        return None
    if year is None:
        year = datetime.datetime.now().year
    return template.format(year=year, author=author)
